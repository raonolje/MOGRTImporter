// (n) TextDecoder utf-16be / utf-16le / euc-kr in the CEP 12 panel (Chromium 99)
await (async () => {
  const out = {};
  const be = new Uint8Array([0xFE, 0xFF, 0xD5, 0x5C, 0xAE, 0x00, 0x00, 0x0A, 0x00, 0x41]); // BOM + "한글\nA"
  const le = new Uint8Array([0xFF, 0xFE, 0x5C, 0xD5, 0x00, 0xAE, 0x0A, 0x00, 0x41, 0x00]);
  try { const d = new TextDecoder("utf-16be"); out.utf16be = JSON.stringify(d.decode(be)); out.utf16beIgnoreBOMdefault = d.ignoreBOM; } catch (e) { out.utf16be = "THROW " + e.message; }
  try { out.utf16be_fatal = JSON.stringify(new TextDecoder("utf-16be", { fatal: true }).decode(be)); } catch (e) { out.utf16be_fatal = "THROW " + e.message; }
  try { out.utf16le = JSON.stringify(new TextDecoder("utf-16le").decode(le)); } catch (e) { out.utf16le = "THROW " + e.message; }
  try { out.eucKr = new TextDecoder("euc-kr").encoding; } catch (e) { out.eucKr = "THROW " + e.message; }
  try { out.cp949 = new TextDecoder("cp949").encoding; } catch (e) { out.cp949 = "THROW " + e.message; }
  // odd byte count in utf-16be (truncated file)
  try { out.utf16be_odd = JSON.stringify(new TextDecoder("utf-16be").decode(be.slice(0, 9))); } catch (e) { out.utf16be_odd = "THROW " + e.message; }
  return JSON.stringify(out);
})()
